def fail(message: str) -> None:
    raise SystemExit(f"Phase 8 postmortem verifier rejected launch: {message}")


def require_real_directory(path: Path, field_name: str) -> os.stat_result:
    try:
        metadata = path.lstat()
    except FileNotFoundError:
        fail(f"{field_name} is absent")
    if stat.S_ISLNK(metadata.st_mode) or not stat.S_ISDIR(metadata.st_mode):
        fail(f"{field_name} is not a real directory")
    return metadata


def require_real_file(path: Path, field_name: str) -> os.stat_result:
    try:
        metadata = path.lstat()
    except FileNotFoundError:
        fail(f"{field_name} is absent")
    if stat.S_ISLNK(metadata.st_mode) or not stat.S_ISREG(metadata.st_mode):
        fail(f"{field_name} is not a real regular file")
    return metadata


def canonical_relative_path(raw: str, field_name: str) -> PurePosixPath:
    path = PurePosixPath(raw)
    if not raw or "\\" in raw or "//" in raw or path.is_absolute():
        fail(f"{field_name} is not canonical")
    if path.as_posix() != raw or any(part in {"", ".", ".."} for part in path.parts):
        fail(f"{field_name} is not canonical")
    return path


def read_regular_repo_file_no_follow(raw: str) -> tuple[bytes, os.stat_result]:
    relative = canonical_relative_path(raw, "tracked path")
    if PINNED_GIT is None:
        directory_fd = os.open(REPO_ROOT, os.O_RDONLY | os.O_DIRECTORY | os.O_NOFOLLOW)
    else:
        directory_fd = os.dup(PINNED_GIT.repo_fd)
    try:
        for part in relative.parts[:-1]:
            next_fd = os.open(part, os.O_RDONLY | os.O_DIRECTORY | os.O_NOFOLLOW, dir_fd=directory_fd)
            os.close(directory_fd)
            directory_fd = next_fd
        file_fd = os.open(relative.parts[-1], os.O_RDONLY | os.O_NOFOLLOW, dir_fd=directory_fd)
        try:
            before = os.fstat(file_fd)
            if not stat.S_ISREG(before.st_mode):
                fail(f"tracked path is not a regular file: {raw}")
            chunks: list[bytes] = []
            while True:
                chunk = os.read(file_fd, 1024 * 1024)
                if not chunk:
                    break
                chunks.append(chunk)
            after = os.fstat(file_fd)
            path_after = os.stat(relative.parts[-1], dir_fd=directory_fd, follow_symlinks=False)
            identity_before = (before.st_dev, before.st_ino, before.st_mode, before.st_size)
            identity_after = (after.st_dev, after.st_ino, after.st_mode, after.st_size)
            identity_path = (path_after.st_dev, path_after.st_ino, path_after.st_mode, path_after.st_size)
            if identity_before != identity_after or identity_before != identity_path:
                fail(f"tracked path changed while captured: {raw}")
            return b"".join(chunks), path_after
        finally:
            os.close(file_fd)
    except (FileNotFoundError, NotADirectoryError, OSError) as exc:
        fail(f"tracked path could not be captured without following links: {raw}: {exc}")
    finally:
        os.close(directory_fd)


def git_blob_oid(payload: bytes) -> str:
    header = f"blob {len(payload)}\0".encode("ascii")
    return hashlib.sha1(header + payload, usedforsecurity=False).hexdigest()


def descriptor_identity(fd: int) -> tuple[int, int, int, int]:
    metadata = os.fstat(fd)
    return (metadata.st_dev, metadata.st_ino, metadata.st_mode, metadata.st_size)


def read_regular_fd(fd: int, field_name: str) -> bytes:
    before = os.fstat(fd)
    if not stat.S_ISREG(before.st_mode):
        fail(f"{field_name} is not a regular file")
    payload = bytearray()
    offset = 0
    while offset < before.st_size:
        chunk = os.pread(fd, min(1024 * 1024, before.st_size - offset), offset)
        if not chunk:
            fail(f"{field_name} was truncated while read")
        payload.extend(chunk)
        offset += len(chunk)
    after = os.fstat(fd)
    if descriptor_identity(fd) != (
        before.st_dev,
        before.st_ino,
        before.st_mode,
        before.st_size,
    ):
        fail(f"{field_name} changed while read")
    return bytes(payload)


def snapshot_directory_tree(root_fd: int, field_name: str) -> tuple[tuple[object, ...], ...]:
    rows: list[tuple[object, ...]] = []
    stack = [(PurePosixPath(), os.dup(root_fd))]
    while stack:
        relative_parent, current_fd = stack.pop()
        try:
            entries = sorted(os.scandir(current_fd), key=lambda entry: entry.name)
        except OSError as exc:
            os.close(current_fd)
            fail(f"{field_name} cannot be enumerated: {exc}")
        try:
            for entry in entries:
                relative = relative_parent / entry.name
                try:
                    metadata = os.stat(entry.name, dir_fd=current_fd, follow_symlinks=False)
                except OSError as exc:
                    fail(f"{field_name} entry cannot be inspected: {relative}: {exc}")
                identity = (metadata.st_dev, metadata.st_ino, metadata.st_mode, metadata.st_size)
                if stat.S_ISLNK(metadata.st_mode):
                    fail(f"{field_name} contains a symlink: {relative}")
                if stat.S_ISDIR(metadata.st_mode):
                    child_fd = os.open(
                        entry.name,
                        os.O_RDONLY | os.O_DIRECTORY | os.O_NOFOLLOW,
                        dir_fd=current_fd,
                    )
                    if descriptor_identity(child_fd) != identity:
                        os.close(child_fd)
                        fail(f"{field_name} directory identity changed: {relative}")
                    rows.append((relative.as_posix(), *identity, None))
                    stack.append((relative, child_fd))
                    continue
                if not stat.S_ISREG(metadata.st_mode):
                    fail(f"{field_name} contains a non-file entry: {relative}")
                file_fd = os.open(entry.name, os.O_RDONLY | os.O_NOFOLLOW, dir_fd=current_fd)
                try:
                    if descriptor_identity(file_fd) != identity:
                        fail(f"{field_name} entry identity changed: {relative}")
                    payload = read_regular_fd(file_fd, f"{field_name} entry {relative}")
                    path_after = os.stat(entry.name, dir_fd=current_fd, follow_symlinks=False)
                    path_identity = (
                        path_after.st_dev,
                        path_after.st_ino,
                        path_after.st_mode,
                        path_after.st_size,
                    )
                    if path_identity != identity:
                        fail(f"{field_name} entry changed while captured: {relative}")
                finally:
                    os.close(file_fd)
                rows.append(
                    (relative.as_posix(), *identity, hashlib.sha256(payload).hexdigest())
                )
        finally:
            os.close(current_fd)
    return tuple(sorted(rows))


class PinnedGitContext:
    def __init__(self) -> None:
        self.repo_fd = os.open(REPO_ROOT, os.O_RDONLY | os.O_DIRECTORY | os.O_NOFOLLOW)
        self.repo_identity = descriptor_identity(self.repo_fd)
        self.git_fd = os.open(".git", os.O_RDONLY | os.O_DIRECTORY | os.O_NOFOLLOW, dir_fd=self.repo_fd)
        self.objects_fd = os.open("objects", os.O_RDONLY | os.O_DIRECTORY | os.O_NOFOLLOW, dir_fd=self.git_fd)
        self.refs_fd = os.open("refs", os.O_RDONLY | os.O_DIRECTORY | os.O_NOFOLLOW, dir_fd=self.git_fd)
        self.info_fd = os.open("info", os.O_RDONLY | os.O_DIRECTORY | os.O_NOFOLLOW, dir_fd=self.git_fd)
        self.head_fd = os.open("HEAD", os.O_RDONLY | os.O_NOFOLLOW, dir_fd=self.git_fd)
        self.index_fd = os.open("index", os.O_RDONLY | os.O_NOFOLLOW, dir_fd=self.git_fd)
        self.config_fd = os.open("config", os.O_RDONLY | os.O_NOFOLLOW, dir_fd=self.git_fd)
        self.exclude_fd = os.open("exclude", os.O_RDONLY | os.O_NOFOLLOW, dir_fd=self.info_fd)
        self.packed_refs_fd = os.open("packed-refs", os.O_RDONLY | os.O_NOFOLLOW, dir_fd=self.git_fd)
        self.bound = (
            (self.repo_fd, None, None, self.repo_identity),
            (self.git_fd, self.repo_fd, ".git", descriptor_identity(self.git_fd)),
            (self.objects_fd, self.git_fd, "objects", descriptor_identity(self.objects_fd)),
            (self.refs_fd, self.git_fd, "refs", descriptor_identity(self.refs_fd)),
            (self.info_fd, self.git_fd, "info", descriptor_identity(self.info_fd)),
            (self.head_fd, self.git_fd, "HEAD", descriptor_identity(self.head_fd)),
            (self.index_fd, self.git_fd, "index", descriptor_identity(self.index_fd)),
            (self.config_fd, self.git_fd, "config", descriptor_identity(self.config_fd)),
            (self.exclude_fd, self.info_fd, "exclude", descriptor_identity(self.exclude_fd)),
            (
                self.packed_refs_fd,
                self.git_fd,
                "packed-refs",
                descriptor_identity(self.packed_refs_fd),
            ),
        )
        for fd, _parent_fd, name, identity in self.bound:
            metadata = os.fstat(fd)
            if name in {None, ".git", "objects", "refs", "info"}:
                if not stat.S_ISDIR(metadata.st_mode):
                    fail("pinned Git directory binding is not a directory")
            elif not stat.S_ISREG(metadata.st_mode):
                fail("pinned Git file binding is not a regular file")
            if stat.S_ISLNK(metadata.st_mode) or descriptor_identity(fd) != identity:
                fail("pinned Git binding is not canonical")
        self.bound_payloads = {
            fd: hashlib.sha256(read_regular_fd(fd, f"pinned Git file {name}")).hexdigest()
            for fd, _parent_fd, name, _identity in self.bound
            if name not in {None, ".git", "objects", "refs", "info"}
        }
        self.directory_snapshots = {
            self.git_fd: snapshot_directory_tree(self.git_fd, "Git control store"),
            self.objects_fd: snapshot_directory_tree(self.objects_fd, "Git object store"),
            self.refs_fd: snapshot_directory_tree(self.refs_fd, "Git refs store"),
            self.info_fd: snapshot_directory_tree(self.info_fd, "Git info store"),
        }
        self.absent_git_entries = ("commondir", "config.worktree")
        self.absent_info_entries = ("attributes",)
        self.revalidate()

    @property
    def pass_fds(self) -> tuple[int, ...]:
        return tuple(fd for fd, _parent_fd, _name, _identity in self.bound)

    def revalidate(self) -> None:
        observed_repo = REPO_ROOT.lstat()
        if (observed_repo.st_dev, observed_repo.st_ino, observed_repo.st_mode, observed_repo.st_size) != self.repo_identity:
            fail("repository root identity changed")
        for fd, parent_fd, name, identity in self.bound:
            if descriptor_identity(fd) != identity:
                fail("pinned Git descriptor identity changed")
            if parent_fd is not None and name is not None:
                try:
                    metadata = os.stat(name, dir_fd=parent_fd, follow_symlinks=False)
                except FileNotFoundError:
                    fail(f"pinned Git path disappeared: {name}")
                path_identity = (metadata.st_dev, metadata.st_ino, metadata.st_mode, metadata.st_size)
                if path_identity != identity:
                    fail(f"pinned Git path identity changed: {name}")
            expected_payload = self.bound_payloads.get(fd)
            if expected_payload is not None:
                observed_payload = hashlib.sha256(
                    read_regular_fd(fd, f"pinned Git file {name}")
                ).hexdigest()
                if observed_payload != expected_payload:
                    fail(f"pinned Git file content changed: {name}")
        for fd, expected in self.directory_snapshots.items():
            if snapshot_directory_tree(fd, "pinned Git directory") != expected:
                fail("pinned Git directory content changed")
        for name in self.absent_git_entries:
            try:
                os.stat(name, dir_fd=self.git_fd, follow_symlinks=False)
            except FileNotFoundError:
                continue
            fail(f"redirectable Git control input appeared: {name}")
        for name in self.absent_info_entries:
            try:
                os.stat(name, dir_fd=self.info_fd, follow_symlinks=False)
            except FileNotFoundError:
                continue
            fail(f"redirectable Git info input appeared: {name}")

    def git_prefix(self) -> list[str]:
        return [
            GIT,
            "--no-pager",
            "--no-replace-objects",
            "--literal-pathspecs",
            f"--git-dir=/proc/self/fd/{self.git_fd}",
            f"--work-tree=/proc/self/fd/{self.repo_fd}",
            "-c",
            "core.fsmonitor=false",
            "-c",
            "core.hooksPath=/dev/null",
            "-c",
            "core.attributesFile=/dev/null",
            "-c",
            "core.excludesFile=/proc/self/fd/" + str(self.exclude_fd),
            "-c",
            "core.pager=cat",
            "-c",
            "pager.rev-parse=false",
            "-c",
            "pager.ls-tree=false",
            "-c",
            "pager.ls-files=false",
            "-c",
            "pager.cat-file=false",
        ]

    def git_environment(self) -> dict[str, str]:
        return {
            **GIT_ENVIRONMENT,
            "GIT_COMMON_DIR": f"/proc/self/fd/{self.git_fd}",
            "GIT_INDEX_FILE": f"/proc/self/fd/{self.index_fd}",
            "GIT_OBJECT_DIRECTORY": f"/proc/self/fd/{self.objects_fd}",
        }


def require_allowed_git_argv(args: tuple[str, ...]) -> None:
    fixed = {
        ("rev-parse", "--show-object-format"),
        ("rev-parse", "--verify", "HEAD^{commit}"),
        ("ls-files", "--stage", "-z"),
        ("ls-files", "-v", "-z"),
        ("ls-files", "--others", "--exclude-standard", "-z"),
        ("ls-files", "--others", "--ignored", "--exclude-standard", "-z"),
    }
    if args in fixed:
        return
    if len(args) == 4 and args[:3] == ("ls-tree", "-rz", "--full-tree") and GIT_SHA_RE.fullmatch(args[3]):
        return
    if len(args) == 3 and args[:2] in {("cat-file", "-t"), ("cat-file", "blob")} and GIT_SHA_RE.fullmatch(args[2]):
        return
    fail("Git plumbing argv is outside the exact allowlist")


def git_run(*args: str) -> bytes:
    if PINNED_GIT is None:
        fail("Git plumbing requested before descriptor binding")
    require_allowed_git_argv(args)
    PINNED_GIT.revalidate()
    completed = subprocess.run(
        [*PINNED_GIT.git_prefix(), *args],
        cwd=f"/proc/self/fd/{PINNED_GIT.repo_fd}",
        env=PINNED_GIT.git_environment(),
        stdin=subprocess.DEVNULL,
        stdout=subprocess.PIPE,
        stderr=subprocess.PIPE,
        check=False,
        close_fds=True,
        pass_fds=PINNED_GIT.pass_fds,
        timeout=30,
    )
    PINNED_GIT.revalidate()
    if completed.returncode != 0 or completed.stderr:
        fail(f"Git plumbing failed: {args[0]}")
    return completed.stdout


def reject_noncanonical_git_entries(root: Path, field_name: str) -> None:
    stack = [root]
    while stack:
        current = stack.pop()
        try:
            entries = list(os.scandir(current))
        except OSError as exc:
            fail(f"{field_name} cannot be enumerated: {exc}")
        for entry in entries:
            try:
                metadata = entry.stat(follow_symlinks=False)
            except OSError as exc:
                fail(f"{field_name} entry cannot be inspected: {entry.name}: {exc}")
            if stat.S_ISLNK(metadata.st_mode):
                fail(f"{field_name} contains a symlink: {entry.path}")
            if stat.S_ISDIR(metadata.st_mode):
                stack.append(Path(entry.path))
            elif not stat.S_ISREG(metadata.st_mode):
                fail(f"{field_name} contains a non-file entry: {entry.path}")


def require_exact_local_config() -> None:
    if PINNED_GIT is None:
        fail("Git config requested before descriptor binding")
    try:
        payload = read_regular_fd(PINNED_GIT.config_fd, "Git local config").decode("utf-8")
    except UnicodeDecodeError:
        fail("Git local config is not UTF-8")
    parser = configparser.RawConfigParser(interpolation=None, strict=True)
    try:
        parser.read_string(payload)
    except configparser.Error as exc:
        fail(f"Git local config is malformed: {exc}")
    expected = {
        "core": {
            "repositoryformatversion": "0",
            "filemode": "true",
            "bare": "false",
            "logallrefupdates": "true",
        },
        'remote "origin"': {
            "url": "git@github.com:johnxye/CapKnow.git",
            "fetch": "+refs/heads/*:refs/remotes/origin/*",
        },
    }
    observed = {
        section: dict(parser.items(section, raw=True))
        for section in parser.sections()
    }
    if parser.defaults() or observed != expected:
        fail("Git local config is outside the exact inert allowlist")


def reject_redirected_git_state() -> None:
    if PINNED_GIT is None:
        fail("Git state requested before descriptor binding")
    PINNED_GIT.revalidate()
    require_real_directory(REPO_ROOT, "repository root")
    require_real_directory(GIT_DIR, "Git directory")
    require_real_directory(GIT_DIR / "objects", "Git object directory")
    require_real_directory(GIT_DIR / "refs", "Git refs directory")
    reject_noncanonical_git_entries(GIT_DIR / "objects", "Git object store")
    reject_noncanonical_git_entries(GIT_DIR / "refs", "Git refs store")
    require_real_file(GIT_DIR / "HEAD", "Git HEAD")
    require_real_file(GIT_DIR / "index", "Git index")
    config_path = GIT_DIR / "config"
    require_real_file(config_path, "Git local config")
    forbidden_paths = (
        GIT_DIR / "commondir",
        GIT_DIR / "config.worktree",
        GIT_DIR / "info" / "grafts",
        GIT_DIR / "objects" / "info" / "alternates",
        GIT_DIR / "objects" / "info" / "http-alternates",
    )
    if any(os.path.lexists(path) for path in forbidden_paths):
        fail("redirected, alternate, or grafted Git state exists")
    replace_root = GIT_DIR / "refs" / "replace"
    if os.path.lexists(replace_root):
        fail("Git replacement refs exist")
    packed_refs = GIT_DIR / "packed-refs"
    if os.path.lexists(packed_refs):
        packed_payload, _metadata = read_regular_repo_file_no_follow(".git/packed-refs")
        if b" refs/replace/" in packed_payload:
            fail("packed Git replacement refs exist")
    pack_root = GIT_DIR / "objects" / "pack"
    if os.path.lexists(pack_root):
        require_real_directory(pack_root, "Git pack directory")
        if any(entry.name.endswith(".promisor") for entry in os.scandir(pack_root)):
            fail("Git promisor packs exist")
    require_exact_local_config()
    PINNED_GIT.revalidate()


def parse_tree(payload: bytes) -> dict[str, tuple[str, str]]:
    rows: dict[str, tuple[str, str]] = {}
    for raw in payload.split(b"\0"):
        if not raw:
            continue
        try:
            metadata, raw_path = raw.split(b"\t", 1)
            mode, object_type, oid = metadata.decode("ascii").split(" ")
            path = raw_path.decode("utf-8")
        except (UnicodeDecodeError, ValueError):
            fail("HEAD tree contains a malformed entry")
        canonical_relative_path(path, "HEAD tree path")
        if object_type != "blob" or mode != "100644" or GIT_SHA_RE.fullmatch(oid) is None:
            fail(f"HEAD tree contains a non-regular or executable entry: {path}")
        if path in rows:
            fail("HEAD tree contains a duplicate path")
        rows[path] = (mode, oid)
    if not rows:
        fail("HEAD tree is empty")
    return rows


def parse_index(payload: bytes) -> dict[str, tuple[str, str]]:
    rows: dict[str, tuple[str, str]] = {}
    for raw in payload.split(b"\0"):
        if not raw:
            continue
        try:
            metadata, raw_path = raw.split(b"\t", 1)
            mode, oid, stage = metadata.decode("ascii").split(" ")
            path = raw_path.decode("utf-8")
        except (UnicodeDecodeError, ValueError):
            fail("Git index contains a malformed entry")
        canonical_relative_path(path, "index path")
        if stage != "0" or mode != "100644" or GIT_SHA_RE.fullmatch(oid) is None:
            fail(f"Git index contains a non-stage-zero regular entry: {path}")
        if path in rows:
            fail("Git index contains a duplicate path")
        rows[path] = (mode, oid)
    return rows


def parse_index_flags(payload: bytes) -> set[str]:
    paths: set[str] = set()
    for raw in payload.split(b"\0"):
        if not raw:
            continue
        if not raw.startswith(b"H "):
            fail("Git index contains assume-unchanged, skip-worktree, or another special flag")
        try:
            path = raw[2:].decode("utf-8")
        except UnicodeDecodeError:
            fail("Git index flag path is not UTF-8")
        canonical_relative_path(path, "index flag path")
        if path in paths:
            fail("Git index flags contain a duplicate path")
        paths.add(path)
    return paths


def parse_path_list(payload: bytes, field_name: str) -> tuple[str, ...]:
    rows: list[str] = []
    for raw in payload.split(b"\0"):
        if not raw:
            continue
        try:
            path = raw.decode("utf-8")
        except UnicodeDecodeError:
            fail(f"{field_name} contains a non-UTF-8 path")
        canonical_relative_path(path, field_name)
        rows.append(path)
    if len(rows) != len(set(rows)):
        fail(f"{field_name} contains duplicate paths")
    return tuple(rows)


def ignored_import_candidate(path: str) -> bool:
    relative = PurePosixPath(path)
    absolute = REPO_ROOT / path
    try:
        metadata = absolute.lstat()
    except FileNotFoundError:
        fail(f"ignored path disappeared during verification: {path}")
    if stat.S_ISLNK(metadata.st_mode):
        return True
    if metadata.st_mode & 0o111:
        return True
    if "__pycache__" in relative.parts or relative.suffix in IGNORED_IMPORT_SUFFIXES:
        return True
    return relative.name in IGNORED_IMPORT_HOOK_STEMS or relative.stem in IGNORED_IMPORT_HOOK_STEMS


def require_regular_untracked_path(path: str) -> None:
    relative = canonical_relative_path(path, "untracked path")
    try:
        metadata = (REPO_ROOT / relative).lstat()
    except FileNotFoundError:
        fail(f"untracked path disappeared during verification: {path}")
    if stat.S_ISLNK(metadata.st_mode) or not stat.S_ISREG(metadata.st_mode):
        fail(f"untracked path is not a regular non-symlink file: {path}")


def authenticate_proposal() -> tuple[str, str, str]:
    if git_run("cat-file", "-t", PROPOSAL_COMMIT) != b"commit\n":
        fail("accepted proposal object is not a commit")
    proposal_tree = parse_tree(
        git_run("ls-tree", "-rz", "--full-tree", PROPOSAL_COMMIT)
    )
    if proposal_tree.get(PROPOSAL_PATH) != ("100644", PROPOSAL_BLOB):
        fail("accepted proposal commit does not bind the frozen proposal path/blob")
    proposal_payload = git_run("cat-file", "blob", PROPOSAL_BLOB)
    if git_blob_oid(proposal_payload) != PROPOSAL_BLOB:
        fail("accepted proposal blob bytes do not match the frozen blob id")
    return PROPOSAL_COMMIT, PROPOSAL_PATH, PROPOSAL_BLOB


def authenticate_repository(
    accepted_commit: str,
) -> tuple[dict[str, bytes], dict[str, int], tuple[str, ...], tuple[str, ...]]:
    reject_redirected_git_state()
    if git_run("rev-parse", "--show-object-format") != b"sha1\n":
        fail("Git object format is not SHA-1")
    head = git_run("rev-parse", "--verify", "HEAD^{commit}").decode("ascii").strip()
    if head != accepted_commit:
        fail("runtime HEAD is not the accepted implementation commit")
    if git_run("cat-file", "-t", accepted_commit) != b"commit\n":
        fail("accepted implementation object is not a commit")
    tree = parse_tree(git_run("ls-tree", "-rz", "--full-tree", accepted_commit))
    index = parse_index(git_run("ls-files", "--stage", "-z"))
    flags = parse_index_flags(git_run("ls-files", "-v", "-z"))
    if index != tree or flags != set(tree):
        fail("HEAD tree, index entries, or index flags differ")

    captured: dict[str, bytes] = {}
    worktree_modes: dict[str, int] = {}
    for path, (_mode, expected_oid) in sorted(tree.items()):
        payload, metadata = read_regular_repo_file_no_follow(path)
        worktree_modes[path] = metadata.st_mode
        if metadata.st_mode & 0o111:
            fail(f"worktree executable bit differs from HEAD: {path}")
        if git_blob_oid(payload) != expected_oid:
            fail(f"worktree bytes differ from accepted HEAD: {path}")
        if path == RUNNER_PATH or (path.startswith("capability_certificate_lab/") and path.endswith(".py")):
            if git_run("cat-file", "blob", expected_oid) != payload:
                fail(f"Git object bytes differ from captured worktree bytes: {path}")
            captured[path] = payload

    untracked = parse_path_list(
        git_run("ls-files", "--others", "--exclude-standard", "-z"),
        "untracked path",
    )
    for path in untracked:
        relative = PurePosixPath(path)
        if relative != ARTIFACT_PREFIX and ARTIFACT_PREFIX not in relative.parents:
            fail(f"untracked path lies outside the bound artifact parent: {path}")
        require_regular_untracked_path(path)

    ignored = parse_path_list(
        git_run("ls-files", "--others", "--ignored", "--exclude-standard", "-z"),
        "ignored path",
    )
    dangerous = [path for path in ignored if ignored_import_candidate(path)]
    if dangerous:
        fail(f"ignored import surface exists: {dangerous[0]}")
    if PINNED_GIT is None:
        fail("repository authentication lost its descriptor binding")
    PINNED_GIT.revalidate()
    return captured, worktree_modes, untracked, ignored


def module_name_for_path(path: str) -> tuple[str, bool]:
    relative = PurePosixPath(path)
    if relative.parts[0] != "capability_certificate_lab" or relative.suffix != ".py":
        fail(f"invalid repository module path: {path}")
    if relative.name == "__init__.py":
        return ".".join(relative.parts[:-1]), True
    return ".".join((*relative.parts[:-1], relative.stem)), False


class RepositoryAuthority:
    __slots__ = (
        "_accepted_commit",
        "_captured",
        "_captured_modes",
        "_proposal_binding",
        "_authenticate_repository",
        "_authenticate_repository_code",
        "_authenticate_proposal",
        "_authenticate_proposal_code",
        "_fail",
        "_fail_code",
        "_call_code",
        "_sealed",
    )

    def __init__(
        self,
        accepted_commit: str,
        captured: MappingProxyType,
        captured_modes: MappingProxyType,
        proposal_binding: tuple[str, str, str],
    ) -> None:
        object.__setattr__(self, "_accepted_commit", accepted_commit)
        object.__setattr__(self, "_captured", captured)
        object.__setattr__(self, "_captured_modes", captured_modes)
        object.__setattr__(self, "_proposal_binding", proposal_binding)
        object.__setattr__(self, "_authenticate_repository", authenticate_repository)
        object.__setattr__(self, "_authenticate_repository_code", authenticate_repository.__code__)
        object.__setattr__(self, "_authenticate_proposal", authenticate_proposal)
        object.__setattr__(self, "_authenticate_proposal_code", authenticate_proposal.__code__)
        object.__setattr__(self, "_fail", fail)
        object.__setattr__(self, "_fail_code", fail.__code__)
        object.__setattr__(self, "_call_code", type(self).__call__.__code__)
        object.__setattr__(self, "_sealed", True)

    def __setattr__(self, name: str, value: object) -> None:
        if getattr(self, "_sealed", False):
            fail("repository authority object is sealed")
        object.__setattr__(self, name, value)

    def __call__(self) -> tuple[tuple[str, ...], tuple[str, ...]]:
        if (
            type(self).__call__.__code__ is not self._call_code
            or authenticate_repository is not self._authenticate_repository
            or authenticate_repository.__code__ is not self._authenticate_repository_code
            or authenticate_proposal is not self._authenticate_proposal
            or authenticate_proposal.__code__ is not self._authenticate_proposal_code
            or fail is not self._fail
            or fail.__code__ is not self._fail_code
        ):
            self._fail("repository authority behavior binding changed")
        current_captured, current_modes, current_untracked, current_ignored = (
            self._authenticate_repository(self._accepted_commit)
        )
        if current_captured != self._captured or current_modes != self._captured_modes:
            self._fail("captured repository sources or modes changed after verifier preflight")
        if self._authenticate_proposal() != self._proposal_binding:
            self._fail("accepted proposal binding changed after verifier preflight")
        if (
            type(self).__call__.__code__ is not self._call_code
            or authenticate_repository is not self._authenticate_repository
            or authenticate_repository.__code__ is not self._authenticate_repository_code
            or authenticate_proposal is not self._authenticate_proposal
            or authenticate_proposal.__code__ is not self._authenticate_proposal_code
            or fail is not self._fail
            or fail.__code__ is not self._fail_code
        ):
            self._fail("repository authority behavior binding changed during callback")
        return current_untracked, current_ignored


class ClosedRepositoryLoader:
    __slots__ = ("_sources", "_sealed")

    def __init__(self, captured: dict[str, bytes]) -> None:
        sources: dict[str, tuple[str, bytes, bool]] = {}
        for path, payload in captured.items():
            if not path.startswith("capability_certificate_lab/") or not path.endswith(".py"):
                continue
            module_name, is_package = module_name_for_path(path)
            if module_name in sources:
                fail(f"duplicate repository module name: {module_name}")
            sources[module_name] = (path, payload, is_package)
        if "capability_certificate_lab" not in sources:
            fail("closed repository package root is absent")
        object.__setattr__(self, "_sources", MappingProxyType(sources))
        object.__setattr__(self, "_sealed", True)

    @property
    def sources(self) -> MappingProxyType:
        return self._sources

    def __setattr__(self, name: str, value: object) -> None:
        if getattr(self, "_sealed", False):
            fail("closed repository loader is sealed")
        object.__setattr__(self, name, value)

    def find_spec(
        self,
        fullname: str,
        path: object = None,
        target: ModuleType | None = None,
    ) -> importlib.machinery.ModuleSpec | None:
        if fullname in self.sources:
            _source_path, _payload, is_package = self.sources[fullname]
            return importlib.util.spec_from_loader(fullname, self, is_package=is_package)
        if fullname == "capability_certificate_lab" or fullname.startswith("capability_certificate_lab."):
            raise ImportError(f"repository module is outside the closed verifier set: {fullname}")
        return None

    def create_module(self, spec: importlib.machinery.ModuleSpec) -> ModuleType | None:
        return None

    def exec_module(self, module: ModuleType) -> None:
        source_path, payload, is_package = self.sources[module.__name__]
        canonical_path = str(REPO_ROOT / source_path)
        module.__file__ = canonical_path
        module.__cached__ = None
        if is_package:
            module.__path__ = []
        code = compile(payload, canonical_path, "exec", dont_inherit=True)
        exec(code, module.__dict__, module.__dict__)


def parse_launch() -> tuple[str, str, str, list[str]]:
    if os.getcwd() != str(REPO_ROOT):
        fail("CWD is not the canonical repository root")
    if os.environ != EXPECTED_ENVIRONMENT:
        fail("process environment is not the exact supervisor-supplied environment")
    if sys.flags.isolated != 1 or sys.flags.dont_write_bytecode != 1 or sys.flags.no_site != 1:
        fail("isolated/no-bytecode/no-site interpreter flags are not active")
    if sys.path != INITIAL_SYS_PATH or "site" in sys.modules:
        fail("initial import path is not the frozen standard-library-only path")
    if len(sys.argv) != 19 or len(sys.orig_argv) != 24:
        fail("verifier argument cardinality is not exact")
    if sys.argv[1] != "--verifier-sha256" or sys.argv[3] != "--accepted-implementation-commit":
        fail("verifier binding arguments are malformed")
    if sys.argv[5] != "--runner-path" or sys.argv[7] != "--":
        fail("runner binding arguments are malformed")
    verifier_sha = sys.argv[2]
    accepted_commit = sys.argv[4]
    runner_path = sys.argv[6]
    runner_args = sys.argv[8:]
    if re.fullmatch(r"[0-9a-f]{64}", verifier_sha) is None:
        fail("verifier SHA-256 is malformed")
    if GIT_SHA_RE.fullmatch(accepted_commit) is None:
        fail("accepted implementation commit is malformed")
    if runner_path != RUNNER_PATH:
        fail("runner path is not canonical")
    expected_runner_args = [
        "postmortem-failure",
        "--device",
        "cuda:0",
        "--input-root",
        INPUT_ROOT,
        "--output-root",
        OUTPUT_ROOT,
        "--accepted-proposal-commit",
        PROPOSAL_COMMIT,
        "--accepted-implementation-commit",
        accepted_commit,
    ]
    expected_verifier_args = [
        "--verifier-sha256",
        verifier_sha,
        "--accepted-implementation-commit",
        accepted_commit,
        "--runner-path",
        RUNNER_PATH,
        "--",
        *expected_runner_args,
    ]
    expected_sys_argv = ["-c", *expected_verifier_args]
    if sys.argv != expected_sys_argv or runner_args != expected_runner_args:
        fail("verifier or runner arguments differ from the frozen command")
    source = sys.orig_argv[5]
    if hashlib.sha256(source.encode("utf-8")).hexdigest() != verifier_sha:
        fail("inline verifier source does not match its SHA-256 argument")
    expected_orig_argv = [PYTHON, "-I", "-B", "-S", "-c", source, *expected_verifier_args]
    if sys.orig_argv != expected_orig_argv:
        fail("kernel Python argv is not exact")
    if os.readlink("/proc/self/exe") != PYTHON_EXE:
        fail("kernel executable is not the frozen Python binary")
    cmdline_fd = os.open("/proc/self/cmdline", os.O_RDONLY | os.O_NOFOLLOW)
    try:
        cmdline = bytearray()
        while True:
            chunk = os.read(cmdline_fd, 1024 * 1024)
            if not chunk:
                break
            cmdline.extend(chunk)
    finally:
        os.close(cmdline_fd)
    expected_cmdline = b"\0".join(arg.encode("utf-8") for arg in expected_orig_argv) + b"\0"
    if bytes(cmdline) != expected_cmdline:
        fail("/proc/self/cmdline is not exact")
    return verifier_sha, accepted_commit, runner_path, runner_args


def main() -> None:
    global PINNED_GIT
    verifier_sha, accepted_commit, runner_path, runner_args = parse_launch()
    PINNED_GIT = PinnedGitContext()
    mutable_captured, mutable_captured_modes, initial_untracked, initial_ignored = authenticate_repository(
        accepted_commit
    )
    captured = MappingProxyType(dict(mutable_captured))
    captured_modes = MappingProxyType(dict(mutable_captured_modes))
    proposal_binding = authenticate_proposal()
    runner_payload = captured.get(runner_path)
    if runner_payload is None:
        fail("accepted runner bytes were not captured")
    if any(
        name == "capability_certificate_lab" or name.startswith("capability_certificate_lab.")
        for name in sys.modules
    ):
        fail("repository package was imported before closed-loader installation")
    require_real_directory(Path(SITE_PACKAGES), "frozen site-packages directory")
    loader = ClosedRepositoryLoader(captured)
    sys.meta_path.insert(0, loader)
    sys.path.append(SITE_PACKAGES)
    sys.argv = [str(REPO_ROOT / runner_path), *runner_args]

    repository_authority = RepositoryAuthority(
        accepted_commit,
        captured,
        captured_modes,
        proposal_binding,
    )
    if repository_authority() != (initial_untracked, initial_ignored):
        fail("repository path status changed after verifier preflight")

    authority_token = object()
    runner_globals = {
        "__name__": "__main__",
        "__file__": str(REPO_ROOT / runner_path),
        "__package__": None,
        "__cached__": None,
        "__builtins__": __builtins__,
        "__phase8_verifier_sha256__": verifier_sha,
        "__phase8_accepted_implementation_commit__": accepted_commit,
        "__phase8_runner_blob__": git_blob_oid(runner_payload),
        "__phase8_proposal_binding__": proposal_binding,
        "__phase8_authority_token__": authority_token,
        "__phase8_repository_loader__": loader,
        "__phase8_verify_repository_unchanged__": repository_authority,
    }
    code = compile(runner_payload, str(REPO_ROOT / runner_path), "exec", dont_inherit=True)
    exec(code, runner_globals, runner_globals)

if __name__ == "__main__":
    main()
